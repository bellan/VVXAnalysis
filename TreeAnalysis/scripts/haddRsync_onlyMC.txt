#!/bin/bash

# Controlla se il path e' stato passato come argomento
if [ -z "$1" ]; then
    echo "Usage: $0 <path_to_myRes>"
    exit 1
fi

myRes=$1
region="SR2P"

# Esegue i comandi hadd

hadd $myRes/2016preVFP/VZGAnalyzer_$region/DYJetsToLL_M50.root results/2016preVFP/VZGAnalyzer_SR2P/DYJetsToLL_M50_part1of3.root results/2016preVFP/VZGAnalyzer_SR2P/DYJetsToLL_M50_part2of3.root results/2016preVFP/VZGAnalyzer_SR2P/DYJetsToLL_M50_part3of3.root
hadd $myRes/2016postVFP/VZGAnalyzer_$region/DYJetsToLL_M50.root results/2016postVFP/VZGAnalyzer_SR2P/DYJetsToLL_M50_part1of2.root results/2016postVFP/VZGAnalyzer_SR2P/DYJetsToLL_M50_part2of2.root
hadd $myRes/2017/VZGAnalyzer_$region/DYJetsToLL_M50.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part1of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part2of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part3of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part4of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part5of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part6of6.root
hadd $myRes/2018/VZGAnalyzer_$region/DYJetsToLL_M50.root results/2018/VZGAnalyzer_SR2P/DYJetsToLL_M50_part1of6.root results/2018/VZGAnalyzer_SR2P/DYJetsToLL_M50_part2of6.root results/2018/VZGAnalyzer_SR2P/DYJetsToLL_M50_part3of6.root results/2018/VZGAnalyzer_SR2P/DYJetsToLL_M50_part4of6.root results/2018/VZGAnalyzer_SR2P/DYJetsToLL_M50_part5of6.root results/2018/VZGAnalyzer_SR2P/DYJetsToLL_M50_part6of6.root

# Esegue i comandi rsync
rsync results/2018/VZGAnalyzer_SR2P/ZGToLLG.root $myRes/2018/VZGAnalyzer_$region/
rsync results/2017/VZGAnalyzer_SR2P/ZGToLLG.root $myRes/2017/VZGAnalyzer_$region/
rsync results/2016preVFP/VZGAnalyzer_SR2P/ZGToLLG.root $myRes/2016preVFP/VZGAnalyzer_$region/
rsync results/2016postVFP/VZGAnalyzer_SR2P/ZGToLLG.root $myRes/2016postVFP/VZGAnalyzer_$region/

rsync results/2016preVFP/VZGAnalyzer_SR2P/TTTo2L2Nu.root $myRes/2016preVFP/VZGAnalyzer_$region/
rsync results/2016postVFP/VZGAnalyzer_SR2P/TTTo2L2Nu.root $myRes/2016postVFP/VZGAnalyzer_$region/

hadd $myRes/2018/VZGAnalyzer_$region/TTTo2L2Nu.root results/2018/VZGAnalyzer_SR2P/TTTo2L2Nu_part1of4.root results/2018/VZGAnalyzer_SR2P/TTTo2L2Nu_part2of4.root results/2018/VZGAnalyzer_SR2P/TTTo2L2Nu_part3of4.root results/2018/VZGAnalyzer_SR2P/TTTo2L2Nu_part4of4.root
hadd $myRes/2017/VZGAnalyzer_$region/TTTo2L2Nu.root results/2017/VZGAnalyzer_SR2P/TTTo2L2Nu_part1of2.root results/2017/VZGAnalyzer_SR2P/TTTo2L2Nu_part2of2.root

hadd $myRes/2016preVFP/VZGAnalyzer_$region/VZG.root results/2016preVFP/VZGAnalyzer_SR2P/WZGTo2L2jG.root results/2016preVFP/VZGAnalyzer_SR2P/ZZGTo2L2jG.root
hadd $myRes/2016postVFP/VZGAnalyzer_$region/VZG.root results/2016postVFP/VZGAnalyzer_SR2P/WZGTo2L2jG.root results/2016postVFP/VZGAnalyzer_SR2P/ZZGTo2L2jG.root
hadd $myRes/2017/VZGAnalyzer_$region/VZG.root results/2017/VZGAnalyzer_SR2P/WZGTo2L2jG.root results/2017/VZGAnalyzer_SR2P/ZZGTo2L2jG.root
hadd $myRes/2018/VZGAnalyzer_$region/VZG.root results/2018/VZGAnalyzer_SR2P/WZGTo2L2jG.root results/2018/VZGAnalyzer_SR2P/ZZGTo2L2jG.root

exit 0
