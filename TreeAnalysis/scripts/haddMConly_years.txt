#!/bin/bash

# Controlla se il path e' stato passato come argomento
if [ -z "$1" ]; then
    echo "Usage: $0 <path_to_myRes>"
    exit 1
fi

myRes=$1
myResFullRun2=$2
region="SR2P"

# Esegue i comandi hadd

hadd $myResFullRun2/Run2/VZGAnalyzer_$region/VZG.root $myRes/2018/VZGAnalyzer_$region/VZG.root  $myRes/2017/VZGAnalyzer_$region/VZG.root  $myRes/2016postVFP/VZGAnalyzer_$region/VZG.root $myRes/2016preVFP/VZGAnalyzer_$region/VZG.root 
hadd $myResFullRun2/Run2/VZGAnalyzer_$region/DYJetsToLL_M50.root $myRes/2018/VZGAnalyzer_$region/DYJetsToLL_M50.root  $myRes/2017/VZGAnalyzer_$region/DYJetsToLL_M50.root  $myRes/2016postVFP/VZGAnalyzer_$region/DYJetsToLL_M50.root  $myRes/2016preVFP/VZGAnalyzer_$region/DYJetsToLL_M50.root  
hadd $myResFullRun2/Run2/VZGAnalyzer_$region/ZGToLLG.root  $myRes/2018/VZGAnalyzer_$region/ZGToLLG.root  $myRes/2017/VZGAnalyzer_$region/ZGToLLG.root  $myRes/2016postVFP/VZGAnalyzer_$region/ZGToLLG.root $myRes/2016preVFP/VZGAnalyzer_$region/ZGToLLG.root  
hadd $myResFullRun2/Run2/VZGAnalyzer_$region/TTTo2L2Nu.root $myRes/2018/VZGAnalyzer_$region/TTTo2L2Nu.root   $myRes/2017/VZGAnalyzer_$region/TTTo2L2Nu.root   $myRes/2016postVFP/VZGAnalyzer_$region/TTTo2L2Nu.root   $myRes/2016preVFP/VZGAnalyzer_$region/TTTo2L2Nu.root   

exit 0
