#!/bin/bash

# Controlla se il path e' stato passato come argomento

    echo "Usage: $0 <path_to_myRes>"
    exit 1

# Esegue i comandi hadd

hadd $myRes/2016preVFP/VZGAnalyzer_$region/data_obs.root results/2016/VZGAnalyzer_SR2P/2016Bver2.root results/2016/VZGAnalyzer_SR2P/2016Chipm.root results/2016/VZGAnalyzer_SR2P/2016Dhipm.root results/2016/VZGAnalyzer_SR2P/2016Ehipm.root results/2016/VZGAnalyzer_SR2P/2016Fhipm.root

hadd $myRes/2017/VZGAnalyzer_$region/DYJetsToLL_M50.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part1of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part2of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part3of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part4of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part5of6.root results/2017/VZGAnalyzer_SR2P/DYJetsToLL_M50_part6of6.root
